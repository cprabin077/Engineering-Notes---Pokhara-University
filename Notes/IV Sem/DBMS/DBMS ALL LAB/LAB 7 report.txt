 create table sailors
    (
    sid int primary key,
    sname varchar(20),
    rating int,
    age int
    );
+--------+-------------+------+-----+---------+-------+
| Field  | Type        | Null | Key | Default | Extra |
+--------+-------------+------+-----+---------+-------+
| sid    | int(11)     | NO   | PRI | NULL    |       |
| sname  | varchar(20) | YES  |     | NULL    |       |
| rating | int(11)     | YES  |     | NULL    |       |
| age    | int(11)     | YES  |     | NULL    |       |
+--------+-------------+------+-----+---------+-------+
create table boats
    (
    bid int primary key,
    bname varchar(20),
     color varchar(20)
     );
+-------+-------------+------+-----+---------+-------+
| Field | Type        | Null | Key | Default | Extra |
+-------+-------------+------+-----+---------+-------+
| bid   | int(11)     | NO   | PRI | NULL    |       |
| bname | varchar(20) | YES  |     | NULL    |       |
| color | varchar(20) | YES  |     | NULL    |       |
+-------+-------------+------+-----+---------+-------+
create table reserves
    (
   rid int primary key,
    sid int,
    bid int,
    day date,
    foreign key(sid) references sailors(sid),
    foreign key(bid) references boats(bid)
    );
 desc reserves;

| sid   | int(11) | YES  | MUL | NULL    |       |
| bid   | int(11) | YES  | MUL | NULL    |       |
| day   | date    | YES  |     | NULL    |       |

1. Find all information of sailors who have reserved boat number 4.

query:
select * from sailors where sid in (select sid from reserves where bid = 4);
+-----+-------+--------+------+
| sid | sname | rating | age  |
+-----+-------+--------+------+
|   2 | Raju  |      4 |   20 |
|   3 | Hari  |      3 |   22 |
+-----+-------+--------+------+

or

select * from sailors where sid =any(select sid from reserves where bid = 4);
+-----+-------+--------+------+
| sid | sname | rating | age  |
+-----+-------+--------+------+
|   2 | Raju  |      4 |   20 |
|   3 | Hari  |      3 |   22 |
+-----+-------+--------+------+

or

select s.sid,sname,rating,age from sailors s, reserves r where s.sid = r.sid and r.bid = 4;
