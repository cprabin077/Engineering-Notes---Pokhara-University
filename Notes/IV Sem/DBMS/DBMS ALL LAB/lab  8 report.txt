create table department
    -> (
    -> dno int primary key,
    -> dname varchar(20)
    -> );

 create table student
    -> (
    -> sid int primary key,
    -> sname varchar(20),
    -> sex varchar(10),
    -> dob date,
    -> dno int,
    -> foreign key(dno) references department(dno)
    -> );
create table faculty
    -> (
    -> fid int primary key,
    -> fname varchar(20),
    -> designation varchar(20),
    -> salary float,
    -> dno int,
    -> foreign key(dno) references department(dno)
    -> );

create table course
    -> (
    -> cid int primary key,
    -> cname varchar(20),
    -> credits int,
    -> dno int,
    -> foreign key(dno) references department(dno)
    -> );

create table hostel
    -> (
    -> hid int primary key,
    -> hname varchar(20),
    -> seats int
    -> );

create table register
    -> (
    -> rid int primary key,
    -> sid int,
    -> cid int,
    -> sem int,
    -> hid int,
    -> foreign key(sid) references student(sid),
    -> foreign key(cid) references course(cid),
    -> foreign key(hid) references hostel(hid)
    -> );

create table teaching
    -> (
    -> tid int primary key,
    -> sid int,
    -> cid int,
    -> sem int,
    -> fid int,
    -> foreign key(sid) references student(sid),
    -> foreign key(cid) references course(cid),
	foreign key(fid) references faculty(fid)
    -> );
