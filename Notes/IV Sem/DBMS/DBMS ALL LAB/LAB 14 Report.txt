Trigeer and assertion
syntax :
 create trigger trigger_name
 {befor | after} {insert | delete | update}
 on table_name
for each row

begin
     declare var datatype;
  --
end

Example:
create table tbl_audit_nast
    -> (old_name varchar(20),
    -> new_name varchar(20),
    -> v_user varchar(20),
    -> acc_date date,
    -> v_action varchar(20)
    -> );

  delimiter $$
 create or replace trigger trg_nast
    -> after insert
    -> on tbl_nast
    -> for each row
    -> begin
    -> declare uname varchar(20);
    -> declare adate date;
    -> select user(),sysdate() into uname,adate from dual;
    -> insert into tbl_audit_nast values ( NULL,    new.name,uname,adate,'insert');
    -> end;
    -> $$

$$$

Q. write an assertion for bank database to ensure the assets value for 
the A branch is equal to sum of all amounts lent by the A branch.

create assertion dhn_ass_chk check
( not exists ( select * from branch where bname = 'A'
               and assets != ( select sum(amount) from loan
  				where bname = 'A')));
